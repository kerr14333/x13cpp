cmake_minimum_required(VERSION 3.16)
project(x13cpp
        VERSION 0.0.0
        DESCRIPTION "Faithful C++17 port of Census X-13ARIMA-SEATS v1.1 b61"
        LANGUAGES CXX)

# ---------------------------------------------------------------------------
# Global language settings: strict C++17, no compiler extensions.
# The port must build on RHEL8 gcc 8.5, so nothing newer than C++17.
# ---------------------------------------------------------------------------
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# ---------------------------------------------------------------------------
# Floating-point strictness. Parity with the Fortran oracle requires that the
# compiler never contract a*b+c into an FMA and never enable fast-math.
# ---------------------------------------------------------------------------
add_library(x13_fp_flags INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang|AppleClang|IntelLLVM")
  target_compile_options(x13_fp_flags INTERFACE
    -ffp-contract=off
    -fno-fast-math)
elseif(MSVC)
  target_compile_options(x13_fp_flags INTERFACE
    /fp:precise
    /fp:except-)
endif()

# Optional gcov/gcovr code-coverage instrumentation (-DX13_COVERAGE=ON). Applied
# to x13core PUBLIC so every executable/test linking it is instrumented too.
option(X13_COVERAGE "Instrument for gcov coverage" OFF)
add_library(x13_coverage INTERFACE)
if(X13_COVERAGE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
  # Keep -O2 (via the RelWithDebInfo build type): the SEATS/root bit-exact goldens
  # are last-ULP-sensitive to optimization level, so -O0 diverges. --coverage adds
  # the gcov instrumentation without changing FP results at a fixed -O2.
  target_compile_options(x13_coverage INTERFACE --coverage)
  target_link_options(x13_coverage INTERFACE --coverage)
endif()

# Optional strict warnings (off by default; enable with -DX13_WARNINGS=ON).
option(X13_WARNINGS "Enable strict warnings" OFF)
add_library(x13_warnings INTERFACE)
if(X13_WARNINGS)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
    target_compile_options(x13_warnings INTERFACE -Wall -Wextra)
  elseif(MSVC)
    target_compile_options(x13_warnings INTERFACE /W4)
  endif()
endif()

# ---------------------------------------------------------------------------
# Core static library. Collects every compiled .cpp under core/src EXCEPT the
# unported/ skeletons (generated by tools/f2skel.py; not build-ready yet).
# ---------------------------------------------------------------------------
file(GLOB_RECURSE X13_CORE_SOURCES CONFIGURE_DEPENDS
     "${CMAKE_CURRENT_SOURCE_DIR}/core/src/*.cpp")
list(FILTER X13_CORE_SOURCES EXCLUDE REGEX "/core/src/unported/")

add_library(x13core STATIC ${X13_CORE_SOURCES})
target_include_directories(x13core PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}/core/include"
  "${CMAKE_CURRENT_SOURCE_DIR}/core/src"
  "${CMAKE_CURRENT_SOURCE_DIR}/core/prm"
  "${CMAKE_CURRENT_SOURCE_DIR}/core/prm/gen")
target_link_libraries(x13core PUBLIC x13_fp_flags x13_warnings x13_coverage)

# ---------------------------------------------------------------------------
# x13parse -- M1 CLI harness: parse a spec file and print the .err channel,
# fatal/ok outcome, and captured key settings (used by the M1 parity gate).
# ---------------------------------------------------------------------------
add_executable(x13parse tools/x13parse.cpp)
target_link_libraries(x13parse PRIVATE x13core)

# ---------------------------------------------------------------------------
# x13run_m2 -- M2 CLI harness: run the pre-model phase and write save tables.
# ---------------------------------------------------------------------------
add_executable(x13run_m2 tools/x13run_m2.cpp)
target_link_libraries(x13run_m2 PRIVATE x13core)

# ---------------------------------------------------------------------------
# x13run_m3 -- M3 CLI harness: parse + pre-model + estimate the regARIMA model,
# printing the estimation results to stdout for the estimation parity gate.
# ---------------------------------------------------------------------------
add_executable(x13run_m3 tools/x13run_m3.cpp)
target_link_libraries(x13run_m3 PRIVATE x13core)

# ---------------------------------------------------------------------------
# x13run_iddiff -- M4 harness: pre-model + drive iddiff (automatic differencing-
# order identification), printing the identified (d,D) for the automdl gate.
# ---------------------------------------------------------------------------
add_executable(x13run_iddiff tools/x13run_iddiff.cpp)
target_link_libraries(x13run_iddiff PRIVATE x13core)

# ---------------------------------------------------------------------------
# x13run_x11 -- M5 harness: parse + assemble the X-11 decomposition spine
# (no-model path), printing the B/C/D save tables to stdout for the X-11 gate.
# ---------------------------------------------------------------------------
add_executable(x13run_x11 tools/x13run_x11.cpp)
target_link_libraries(x13run_x11 PRIVATE x13core)

# ---------------------------------------------------------------------------
# Unit tests (self-contained microtest framework; no external dependencies).
# ---------------------------------------------------------------------------
option(X13_BUILD_TESTS "Build unit tests" ON)
if(X13_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests/unit)
endif()
